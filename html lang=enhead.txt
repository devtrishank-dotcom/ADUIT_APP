    /* --- rbTabs: shared ?tab= deep-linking for the hub tab bars ---------------------------
       The hubs each grew their own tab switcher, and none of them put the active tab in the
       URL, so a tab could not be linked to, bookmarked, or survive a refresh. This is the one
       place that owns that: hubs keep their existing switcher and hand it over as render().

       rbTabs({ valid, default, render }) reads ?tab= on load, calls render() with whatever it
       finds (falling back to default when the slug is absent or unrecognised), and returns a
       go(key) that switches and pushState's. Back/forward is wired to the same render, so the
       history entries a user creates by clicking tabs actually work.

       Only for hubs that render every pane into the DOM. Contact Directory and Business Reports
       server-render a single pane, so their tabs stay real navigations — the URL is still
       ?tab=, it is just the server rather than this helper that reads it. */
    window.rbTabs = function (opts) {
        var valid = opts.valid || [];
        var fallback = opts.default || valid[0];

        function fromUrl() {
            var t = new URLSearchParams(window.location.search).get('tab');
            return valid.indexOf(t) !== -1 ? t : fallback;
        }

        /* Build the canonical URL for a tab. opts.base (optional) pins the path, so a hub that
           also answers on a legacy /hub/{tab} path does not end up advertising a contradictory
           '/settings/team?tab=notifications'. Other query params are carried over either way. */
        function urlFor(key) {
            var u = new URL(window.location.href);
            if (opts.base) { u.pathname = new URL(opts.base, window.location.origin).pathname; }
            u.searchParams.set('tab', key);
            return u;
        }

        function apply(key, push) {
            if (valid.indexOf(key) === -1) key = fallback;
            opts.render(key);
            /* A table in a just-revealed pane has only now been laid out, so this is the first
               moment it can be measured for stacking. Nothing else fires here — revealing a pane
               is a class toggle, not a DOM mutation and not a resize. */
            if (window.rbFitTables) window.rbFitTables();
            if (push) {
                try { window.history.pushState({ rbTab: key }, '', urlFor(key)); } catch (e) {}
            }
        }

        window.addEventListener('popstate', function () { apply(fromUrl(), false); });

        /* Apply on load without pushing: the server already rendered a tab, and this either
           agrees with it or corrects it when the server had no ?tab= to read. */
        apply(fromUrl(), false);

        /* Landed on the legacy path form (/settings/team with no ?tab=)? Rewrite the address bar
           to the canonical one in place. replaceState, not push: the user did not navigate, and
           a history entry here would make Back a no-op that looks broken. */
        if (opts.base && !new URLSearchParams(window.location.search).get('tab')) {
            try { window.history.replaceState({ rbTab: fallback }, '', urlFor(fallback)); } catch (e) {}
        }

        return function (key) { apply(key, true); };
    };

       max-width:639px rules render beside each value. Doing it here rather than in markup keeps

       Cells that already carry a data-label are left alone, so a table can override a header

    window.rbStackLabels = function (root) {
        (root || document).querySelectorAll('table.rb-stack').forEach(function (t) {
            /* Mark the scroll wrapper so rbFitTables can find it, and drop any cached natural
               width — an AJAX re-render can change how wide the table wants to be. */

    /* --- rbFitTables: stack a table only when it genuinely does not fit -------------------
       Decides per table rather than per breakpoint, because the space a table has depends on
       the sidebar (300px lg / 230px md, and collapsible) as much as on the window. A table is
       measured once at its natural width, then stacked whenever its wrapper is narrower.

       No feedback loop: the observed element is the WRAPPER, whose width comes from its parent
       block, not from the table inside it — so stacking the table cannot resize what we watch. */
    window.rbFitTables = function () {
        document.querySelectorAll('table.rb-stack').forEach(function (t) {
            var wrap = t.parentElement;
            if (!wrap) return;

            /* A table inside a hidden tab pane measures 0 in both directions. Skip it — caching a
               0 here would mark it "fits" forever, which is what left the Automations and
               Responses tables still dragging sideways after their tab was opened. */
            if (!wrap.clientWidth) return;

            if (!t.dataset.rbNatural) {
                /* Measure unstacked and unconstrained. max-content asks the table how wide it
                   actually wants to be, which is not the same as how wide it currently is —
                   a w-full table always reports its container's width otherwise. */
                var wasStacked = t.classList.contains('rb-stacked');
                t.classList.remove('rb-stacked');
                var prevWidth = t.style.width;
                t.style.width = 'max-content';
                var natural = t.scrollWidth;
                t.style.width = prevWidth;
                if (wasStacked) t.classList.add('rb-stacked');
                if (!natural) return;                       // still not laid out; try again later
                t.dataset.rbNatural = String(natural);
            }

            var needs = parseInt(t.dataset.rbNatural, 10) || 0;
            var stack = needs > wrap.clientWidth + 1;
            t.classList.toggle('rb-stacked', stack);
            wrap.classList.toggle('is-stacked', stack);
        });
    };

<script>
    (function () {
        var panel = document.getElementById('rb-guide');
        if (!panel) return;

        var titleEl = document.getElementById('rb-guide-title');
        var bodyEl  = document.getElementById('rb-guide-body');
        var timer   = null;

        function show(el) {
            titleEl.textContent = el.getAttribute('data-guide-title') || '';
            bodyEl.textContent  = el.getAttribute('data-guide') || '';
            panel.classList.toggle('is-locked', el.hasAttribute('data-guide-locked'));

            // Measure only once the text is in place, otherwise the height is stale and a
            // panel near the bottom of the screen gets clamped against the wrong number.
            var r = el.getBoundingClientRect();
            panel.classList.add('is-visible');
            var h = panel.offsetHeight;

            var top = r.top + (r.height / 2) - (h / 2);
            top = Math.max(8, Math.min(top, window.innerHeight - h - 8));

            // Sit to the right of the rail, or flip to the left if that would overflow —
            // which is what happens on a narrow window with the sidebar expanded.
            var left = r.right + 12;
            if (left + panel.offsetWidth > window.innerWidth - 8) {
                left = Math.max(8, r.left - panel.offsetWidth - 12);
            }

            panel.style.top  = top + 'px';
            panel.style.left = left + 'px';
            panel.setAttribute('aria-hidden', 'false');
        }

        function hide() {
            clearTimeout(timer);
            panel.classList.remove('is-visible');
            panel.setAttribute('aria-hidden', 'true');
        }

        // Delegated so it survives any re-render of the nav, and scoped to [data-guide]
        // so only items that actually carry a description ever open one.
        document.addEventListener('mouseover', function (e) {
            var el = e.target.closest ? e.target.closest('[data-guide]') : null;
            if (!el) return;
            clearTimeout(timer);
            timer = setTimeout(function () { show(el); }, 280);
        });

        document.addEventListener('mouseout', function (e) {
            var el = e.target.closest ? e.target.closest('[data-guide]') : null;
            if (!el) return;
            // Crossing between children of the same item (icon → label) also fires mouseout.
            // Treating that as a leave makes the panel flicker on the way in, so only hide
            // when the cursor has actually left the item.
            if (e.relatedTarget && el.contains(e.relatedTarget)) return;
            hide();
        });

        // Keyboard parity: tabbing through the nav surfaces the same text.
        document.addEventListener('focusin', function (e) {
            var el = e.target.closest ? e.target.closest('[data-guide]') : null;
            if (el) show(el);
        });
        document.addEventListener('focusout', hide);

        // The panel is positioned in viewport coordinates, so any scroll invalidates it.
        document.addEventListener('scroll', hide, true);
        window.addEventListener('resize', hide);
    })();
</script>

        function escapeHtml(s) {

            }[c]));

            let s = escapeHtml(text || '');

            s = s.replace(/(^|\n)(\d+)\.\s+(.+)/g, '$1<div class="flex gap-2 mt-1"><span class="text-violet-500 shrink-0">$2.</span><span>$3</span></div>');
            // Convert remaining \n to <br>

            if(!t) return 'Just now';
            // epoch in seconds
            if(typeof t === 'number'){
                const ms = t < 2e12 ? t*1000 : t; // handle sec vs ms
                return new Date(ms).toLocaleString();
            }
            // numeric string?
            if(/^\d+$/.test(String(t))){
                const num = Number(t);
                const ms = num < 2e12 ? num*1000 : num;
                return new Date(ms).toLocaleString();
            }
            // ISO / parseable

    function syncWatcher({
        establishmentId,
        pollEveryMs = 3000,
        endpoint = '/establishments',
        maxSilentTicks = 3, // how many empty responses before hiding
    }) {
        return {
            visible: true,
            status: 'pending',
            progress: 0,
            total: null,
            processed: 0,
            error: null,
            title: 'Importing your Google reviews…',
            timer: null,
            silentTicks: 0,
            canceling: false,

            get statusLabel() {
                return this.status === 'pending' ? 'Queued' :
                    this.status === 'syncing' ? 'Syncing' :
                    this.status === 'done' ? 'Complete' :
                    'Failed';
            },

            start() {
                // If we don't have an ID, there's nothing to poll — hide immediately
                if (!establishmentId) {
                    this.hide();
                    return;
                }
                this.tick(true);
                this.timer = setInterval(() => this.tick(false), pollEveryMs);
            },

            async tick(force = false) {
                try {
                    const base = (window.api_url || '').replace(/\/+$/, ''); // optional global
                    const ep = endpoint.replace(/^\/?/, '/').replace(/\/+$/, ''); // normalize
                    const url = `${base}${ep}/${encodeURIComponent(establishmentId)}/sync-status`;

                    const res = await fetch(url, {
                        headers: {
                            'Accept': 'application/json'
                        }
                    });

                    // If server says "not found", hide the banner
                    if (res.status === 404) {
                        this.title = 'Nothing to import';
                        this.hide();
                        return;
                    }

                    if (!res.ok) throw new Error(`Status check failed (${res.status})`);

                    const j = await res.json();

                    // If payload is empty/missing expected fields, count a silent tick
                    const hasUsefulData = (
                        j && (
                            j.sync_status !== undefined ||
                            j.sync_progress !== undefined ||
                            j.reviews_total !== undefined ||
                            j.total_review !== undefined ||
                            j.reviews_processed !== undefined
                        )
                    );

                    if (!hasUsefulData) {
                        this.silentTicks++;
                        if (this.silentTicks >= maxSilentTicks) this.hide();
                        return;
                    } else {
                        this.silentTicks = 0;
                    }

                    this.status = j.sync_status ?? 'pending';
                    this.progress = Number(j.sync_progress ?? 0);
                    this.total = j.reviews_total ?? j.total_review ?? null;
                    this.processed = j.reviews_processed ?? 0;
                    this.error = j.last_sync_error ?? null;

                    // Auto-hide when done and 100%
                    if (this.status === 'done' && this.progress >= 100) {
                        this.title = 'Google reviews imported successfully!';
                        clearInterval(this.timer);
                        setTimeout(() => this.hide(), 1200);
                    }

                    // Show failure clearly
                    if (this.status === 'failed') {
                        clearInterval(this.timer);
                        this.title = 'Import failed';
                    }
                } catch (e) {
                    // Network/other errors: count as silent tick; hide after threshold
                    console.warn(e);
                    this.silentTicks++;
                    if (this.silentTicks >= maxSilentTicks) this.hide();
                }
            },

                    console.error(e);

                    this.canceling = false;
                }

        async function initMap() {
            const input = document.getElementById('pac-input');
            const resultsContainer = document.getElementById('search-results');
            const resultsList = document.getElementById('results-list');
            const connectButton = document.getElementById('connect-business');

            // Try new Places JS first
            try {

                map = new Map(document.createElement('div'), {
                    center: biasCenter,
                    zoom: 11,
                });

                useNewPlacesAPI = true;
                initializeNewPlacesAPI(input, resultsContainer, resultsList, connectButton, Place);
            } catch (error) {
                console.log('New Places API not available, using Autocomplete:', error);
                useNewPlacesAPI = false;
                initializeAutocompleteAPI(input, resultsContainer, resultsList, connectButton);
            }

            // Update biasCenter using user's location
            if (navigator.geolocation) {
                navigator.geolocation.getCurrentPosition((position) => {
                    biasCenter = {
                        lat: position.coords.latitude,
                        lng: position.coords.longitude
                    };
                    if (map && map.setCenter) map.setCenter(biasCenter);
                });
            }

        // ==========================================================
        // ✅ New Places JS API (Place.searchByText)
        // with SMART bias + SAB fallback
        // ==========================================================
        function initializeNewPlacesAPI(input, resultsContainer, resultsList, connectButton, Place) {

            async function trySABFallback(query) {
                const sabPlaces = await searchTextSABFallbackREST(query);
                if (!sabPlaces.length) return false;

                renderPlacesList(sabPlaces, resultsList, resultsContainer, async (p) => {
                    await selectPlaceByIdREST(p.id, input, resultsContainer, connectButton, p);
                });
                return true;
            }

            async function findPlaces(query) {
                if (!query || query.trim() === '') {
                    resultsContainer.classList.add('hidden');
                    return;
                }

                setResultsLoading(resultsList, resultsContainer);

                try {
                    // ✅ SMART BIAS:
                    // - If query is generic: bias near user
                    // - If query is specific: do NOT bias (global)
                    const specific = isSpecificQuery(query);

                    const request = {
                        textQuery: query,
                        maxResultCount: 8,
                        language: 'en-US',
                        fields: ['id', 'displayName', 'formattedAddress', 'location', 'businessStatus'],
                    };

                    // Apply bias only for generic queries
                    if (!specific) {
                        request.locationBias = new google.maps.Circle({
                            center: biasCenter,
                            radius: biasRadiusMeters
                        });
                    }

                    const resp = await Place.searchByText(request);
                    const places = resp?.places || [];

                    if (places.length > 0) {
                        renderPlacesList(places, resultsList, resultsContainer, async (p) => {
                            await selectPlaceNew(p, input, resultsContainer, connectButton);
                        });
                    } else {
                        // ✅ if none, try SAB fallback (REST)
                        const sabRendered = await trySABFallback(query);
                        if (!sabRendered) renderNoResults(resultsList, resultsContainer);
                    }

                } catch (err) {
                    console.error("New Places search failed:", err);

                    // Try SAB fallback first
                    try {
                        const sabRendered = await trySABFallback(query);
                        if (!sabRendered) renderNoResults(resultsList, resultsContainer);
                    } catch (e) {
                        renderNoResults(resultsList, resultsContainer);
                    }
                }
            }

            async function selectPlaceNew(place, input, resultsContainer, connectButton) {
                try {
                    // Store name before fetching (in case fetchFields fails or doesn't return displayName)
                    const initialName = getBusinessName(place);
                    
                    // ✅ fetch more details on the Place instance
                    await place.fetchFields({
                        fields: ['id', 'displayName', 'formattedAddress', 'nationalPhoneNumber', 'internationalPhoneNumber']
                    });

                    place_id = place.id;
                    // Use helper function with proper fallbacks
                    business_name = getBusinessName(place) || initialName;
                    business_address = place.formattedAddress || "Service Area Business (address hidden)";
                    business_phone = place.nationalPhoneNumber || place.internationalPhoneNumber || "N/A";
                    icon = "N/A";

                    input.value = business_name;

                    resultsContainer.classList.add('hidden');
                    connectButton.disabled = false;
                    connectButton.classList.remove('opacity-50', 'cursor-not-allowed');

                    selectedPlace = place;
                } catch (e) {
                    console.error("fetchFields failed:", e);
                    // Even if fetchFields fails, use the initial name we stored
                    place_id = place.id || '';
                    business_name = getBusinessName(place);
                    business_address = place.formattedAddress || "Service Area Business (address hidden)";
                    business_phone = "N/A";
                    icon = "N/A";
                    
                    input.value = business_name;
                    resultsContainer.classList.add('hidden');
                    connectButton.disabled = false;
                    connectButton.classList.remove('opacity-50', 'cursor-not-allowed');
                    selectedPlace = place;
                }
            }

            input.addEventListener('keydown', (event) => {
                if (event.key === 'Enter') {
                    event.preventDefault();
                    findPlaces(input.value);
                }
            });

            let searchTimeout;

                clearTimeout(searchTimeout);
                searchTimeout = setTimeout(() => {

                }, 400);
            });

            window.findPlaces = findPlaces;
        }

            function findPlacesAutocomplete(query) {
                if (!query || query.trim() === '') {
                    resultsContainer.classList.add('hidden');
                    return;
                }

                setResultsLoading(resultsList, resultsContainer);

                const autocompleteService = new google.maps.places.AutocompleteService();

                // ⚠️ Autocomplete is also location biased; we keep it biased because it's "near-me" UX.
                autocompleteService.getPlacePredictions({
                    input: query,
                    types: ['establishment'],
                    location: new google.maps.LatLng(biasCenter.lat, biasCenter.lng),
                    radius: biasRadiusMeters
                }, async (predictions, status) => {

                    if (status === google.maps.places.PlacesServiceStatus.OK && predictions && predictions.length) {
                        resultsList.innerHTML = '';
                        resultsContainer.classList.remove('hidden');

                        predictions.slice(0, 8).forEach((prediction) => {
                            const li = document.createElement('li');
                            li.className = 'px-4 py-3 hover:bg-[#F7F9FC] cursor-pointer border-b border-[#DFE2E4A6] last:border-b-0';

                            li.appendChild(nameDiv);

                            li.addEventListener('click', (e) => {
                                e.preventDefault();
                                e.stopPropagation();

                service.getDetails({
                                    placeId: prediction.place_id,
                                    fields: ['place_id', 'name', 'formatted_address', 'formatted_phone_number', 'icon']
                                }, (place, st) => {
                                    if (st === google.maps.places.PlacesServiceStatus.OK) {
                                        place_id = place.place_id;
                                        // Use name from place details, fallback to prediction description if not available
                                        business_name = place.name || prediction.description || "N/A";
                                        business_address = place.formatted_address || "N/A";
                                        business_phone = place.formatted_phone_number || "N/A";
                                        icon = place.icon || "N/A";

                                        // Use business_name variable to ensure we have a value
                                        input.value = business_name;

                                        resultsContainer.classList.add('hidden');
                                        connectButton.disabled = false;
                                        connectButton.classList.remove('opacity-50', 'cursor-not-allowed');

                                        selectedPlace = place;
                                    } else {
                                        alert('Error loading place details. Please try again.');
                                    }
                                });
                            });

                            resultsList.appendChild(li);
                        });

                    } else {
                        // ✅ If autocomplete gives nothing, try SAB fallback (REST)
                        try {
                            const sabRendered = await trySABFallback(query);
                            if (!sabRendered) renderNoResults(resultsList, resultsContainer);
                        } catch (e) {
                            renderNoResults(resultsList, resultsContainer);
                        }
                    }
                });
            }

    function connectPlatform(slug) {
        const url = base_url + `oauth/redirect/${slug}`;

        // Wix finishes its install on Wix's own site and can no longer redirect back to us
        // (modern Wix apps have no registered redirect URL), so sending this tab there would
        // strand the user. Open it alongside instead and keep Connections in view — the
        // install webhook completes the connection, so a refresh here shows it as connected.
        // Shopify's OAuth URL is per-store, so we need the merchant's domain before we can
        // build it. (It used to be hardcoded to one test store, which is why this never
        // prompted before.) Ask, normalise, then hand off.
        if (slug === 'shopify') {
            Swal.fire({
                title: 'Connect your Shopify store',
                input: 'text',
                inputLabel: 'Your store domain',
                inputPlaceholder: 'your-store.myshopify.com',
                confirmButtonText: 'Connect',
                showCancelButton: true,
                inputValidator: (value) => {
                    const shop = (value || '').trim().toLowerCase()
                        .replace(/^https?:\/\//, '').replace(/\/.*$/, '');
                    if (!/^[a-z0-9][a-z0-9-]*\.myshopify\.com$/.test(shop)) {
                        return 'Enter the full domain, e.g. your-store.myshopify.com';
                    }
                },
            }).then((result) => {
                if (!result.isConfirmed) return;
                const shop = result.value.trim().toLowerCase()
                    .replace(/^https?:\/\//, '').replace(/\/.*$/, '');
                window.location.href = url + '?shop=' + encodeURIComponent(shop);
            });
            return;
        }

        if (slug === 'wix') {
            window.open(url, '_blank', 'noopener');
            if (typeof showInfoPopup === 'function') {
                showInfoPopup(
                    'Finish the install in the new tab',
                    'Approve ReviewBoost on your Wix site, then come back here and refresh — Wix will be connected.',
                    'Got it'
                );
            }
            return;
        }

        window.location.href = url;
    }

    $(document).ready(function() {
        let sidebarCollapsed = false;

        // Apply a collapsed state to the sidebar + main content in one place.
        function applyCollapsed(collapsed) {
            $('#sidebar').toggleClass('collapsed', collapsed);
            $('#sidebar-toggle-icon').toggleClass('rotate-180', collapsed);
            $('#hamburger-icon').toggleClass('rotate-90', collapsed);
            $('main').toggleClass('sidebar-collapsed', collapsed);
            sidebarCollapsed = collapsed;
        }

        // Initial state (desktop only — on mobile it's an off-canvas overlay). The sidebar stays
        // OPEN unless the user has explicitly collapsed it with the toggle; that choice persists
        // in localStorage. It previously auto-collapsed on every page except the dashboard, which
        // meant the nav closed itself as soon as you went anywhere — removed at the client's
        // request: "we should keep it opened".
        // The key is versioned deliberately. The previous behaviour wrote '1' on EVERY sidebar
        // click, so essentially every existing user has "collapsed" stored — reading the old key
        // would keep giving them the rail and make this change look like it never shipped. A new
        // key retires those values and lets everyone start open.
        try {
            if (localStorage.getItem('rb_sidebar_collapsed_v2') === '1' && window.innerWidth >= 768) {
                applyCollapsed(true);
            }
            localStorage.removeItem('rb_sidebar_collapsed');
        } catch (e) {}

        // Function to toggle sidebar
        function toggleSidebar() {
            applyCollapsed(!sidebarCollapsed);
            try { localStorage.setItem('rb_sidebar_collapsed_v2', sidebarCollapsed ? '1' : '0'); } catch (e) {}
        }

        // Sidebar toggle button (in sidebar)
        $('#sidebar-toggle-desktop').on('click', function(e) {
            e.preventDefault();
            e.stopPropagation();
            toggleSidebar();
        });

        // Header hamburger menu button (Google Ads style)
        $('#sidebar-toggle-header').on('click', function(e) {
            e.preventDefault();
            e.stopPropagation();
            toggleSidebar();
        });

        // Navigating closes the MOBILE drawer only — it overlays the page, so it has to get out
        // of the way once you've chosen where to go. On desktop the sidebar now stays exactly as
        // the user left it; it used to collapse on every navigation, which is what the client
        // asked to have removed.
        $(document).on('click', '#sidebar a[href]', function() {
            var href = $(this).attr('href') || '';
            // Ignore anchors and JS hooks — those do not navigate, so nothing should close.
            if (!href || href.charAt(0) === '#' || href.indexOf('javascript:') === 0) return;

            if (window.innerWidth < 768) {
                var menuOpen = document.getElementById('menu-open');
                if (menuOpen) menuOpen.checked = false;
            }
        });

        // Handle submenu behavior when sidebar is collapsed
        $(document).on('click', '#sidebar.collapsed .nav-item[data-collapse-toggle]', function(e) {
            e.preventDefault();
            e.stopPropagation();
            // Don't toggle submenus when sidebar is collapsed
            return false;
        });

        // Handle submenu behavior when sidebar is expanded
        $(document).on('click', '#sidebar:not(.collapsed) .nav-item[data-collapse-toggle]', function(e) {
            const target = $(this).attr('data-collapse-toggle');
            const dropdown = $('#' + target);

            if (dropdown.length) {
                dropdown.toggleClass('hidden');
                const isExpanded = !dropdown.hasClass('hidden');
                $(this).attr('aria-expanded', isExpanded);
            }
        });

<script>
    document.addEventListener('alpine:init', () => {
        Alpine.store('modals', {
            openModals: {},
            show(name) {
                this.openModals[name] = true;
                this.lockScroll();
            },
            hide(name) {
                this.openModals[name] = false;
                this.lockScroll();
            },
            isOpen(name) {
                return !!this.openModals[name];
            },
            lockScroll() {
                const isAnyOpen = Object.values(this.openModals).some(Boolean);
                if (isAnyOpen) {
                    document.body.classList.add('overflow-hidden');
                } else {
                    document.body.classList.remove('overflow-hidden');
                }
            }
        });
    });
</script>

    document.addEventListener("DOMContentLoaded", function () {
        // keep the same upload flow as your existing modal
        const fileInput = document.getElementById("thumbnail");
        const imagePreviewContainer = document.getElementById("imagePreviewContainer");
        const imagePreview = document.getElementById("imagePreview");

        fileInput.addEventListener("change", function (event) {
            const file = event.target.files[0];
            if (file) {
                const reader = new FileReader();
                reader.onload = function (e) {
                    imagePreview.src = e.target.result;
                    imagePreviewContainer.classList.remove("hidden");
                    // When the user had no photo, the <img> starts hidden and an initial-letter
                    // avatar stands in its place. Reveal the image and drop the placeholder, or
                    // the chosen photo would be loaded but invisible behind the initial.
                    imagePreview.classList.remove("hidden");
                    var fb = document.getElementById("imagePreviewFallback");
                    if (fb) fb.remove();
                };
                reader.readAsDataURL(file);
            }
        });

        // same API call as the update profile page
        document.getElementById('user-save-btn-modal').addEventListener('click', function (e) {
            e.preventDefault();
            const fd = new FormData(document.getElementById('account_settings_modal'));
            do_formData_ajax({
                api_hook: 'users/account-setting-save',
                data: fd,

            });
        });
    });
